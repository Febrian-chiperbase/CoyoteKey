# Authentication Credentials Example
# Multiple formats supported

# Username:Password format
admin:admin123
user:password
test:test123
demo:demo456
guest:guest789

# Key=Value format for complex credentials
username=admin,password=secret123,scope=read
username=api_user,password=api_pass,client_id=app123,client_secret=secret456
username=oauth_user,password=oauth_pass,scope=read write,grant_type=password

# API Keys and Tokens
api_key=sk_test_1234567890abcdef
api_key=pk_live_abcdef1234567890
token=eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.example.token
bearer_token=ghp_1234567890abcdefghijklmnopqrstuvwxyz

# OAuth Client Credentials
client_id=app_12345,client_secret=secret_abcdef,grant_type=client_credentials
client_id=mobile_app,client_secret=mobile_secret,scope=api:read api:write

# Session Tokens
session_token=sess_1234567890abcdef
session_id=JSESSIONID_1234567890
auth_token=auth_abcdef1234567890

# Custom Authentication
custom_key=custom_value_123
x_api_key=xapi_1234567890
authorization_key=authkey_abcdef123

# Multi-factor Authentication
username=mfa_user,password=mfa_pass,totp_code=123456
username=2fa_user,password=2fa_pass,backup_code=abcdef123

# JWT Tokens (examples)
jwt_token=eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIiOiIxMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoxNTE2MjM5MDIyfQ.SflKxwRJSMeKKF2QT4fwpMeJf36POk6yJV_adQssw5c

# Refresh Tokens

token=bearer_token_123,refresh=refresh_token_456,scope=full_access
